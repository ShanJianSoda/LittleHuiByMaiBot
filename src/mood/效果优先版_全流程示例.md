### 2.2 词典格式转换

将 NRC-VAD 转为 [-1, 1] 区间，便于与文档中的公式一致：

```python
# scripts/convert_vad_to_internal.py
def convert_vad_file(input_path: str, output_path: str):
    """
    NRC-VAD 原值范围 [0,1]，转换为 [-1,1]
    v_new = 2 * v_old - 1
    """
    result = {}
    with open(input_path, "r", encoding="utf-8") as f:
        next(f)  # skip header
        for line in f:
            parts = line.strip().split()
            if len(parts) >= 4:
                word, v, a, d = parts[0], float(parts[1]), float(parts[2]), float(parts[3])
                # 转为 [-1, 1]
                result[word] = {"v": 2*v-1, "a": 2*a-1}
    
    with open(output_path, "w", encoding="utf-8") as f:
        import json
        json.dump(result, f, ensure_ascii=False, indent=0)
    
    print(f"转换完成，共 {len(result)} 词条")
```

**训练部分说明**：效果优先版不做模型训练，仅使用词典，因此「训练」等价于「数据准备 + 词典加载」。

## 四、阶段 3：情绪估计（感知层）

```python
# src/mood/mood_estimator.py
import jieba
from typing import Tuple, Optional
from .vad_lexicon import get_lexicon

def estimate_from_lexicon(text: str) -> Tuple[float, float, float]:
    """
    使用词典估计文本情绪
    Returns: (valence, arousal, confidence)
    """
    if not text or not text.strip():
        return (0.0, 0.0, 0.0)
    
    lexicon = get_lexicon()
    words = list(jieba.cut(text.strip()))
    
    total_v = 0.0
    total_a = 0.0
    matched = 0
    
    for w in words:
        w = w.strip()
        if not w or len(w) < 2:  # 过滤单字、标点
            continue
        if w in lexicon:
            total_v += lexicon[w]["v"]
            total_a += lexicon[w]["a"]
            matched += 1
    
    if matched == 0:
        return (0.0, 0.0, 0.0)
    
    v = total_v / matched
    a = total_a / matched
    # 词数越多，置信度越高，上限 1.0
    confidence = min(1.0, matched / 10.0)
    
    return (round(v, 4), round(a, 4), round(confidence, 4))


# 示例运行
if __name__ == "__main__":
    # 假设词典已加载
    texts = [
        "今天好开心啊，大家一起出去玩！",
        "烦死了，又加班",
        "嗯",
    ]
    for t in texts:
        v, a, conf = estimate_from_lexicon(t)
        print(f"文本: {t[:20]}... -> v={v}, a={a}, conf={conf}")
```

**示例输出**：
```

文本: 烦死了，又加班... -> v=-0.38, a=0.25, conf=0.6
文本: 嗯... -> v=0.0, a=0.0, conf=0.0
```

## 五、阶段 4：动力学更新

```python
# src/mood/mood_dynamics.py
import time
import numpy as np
from typing import Tuple

class MoodDynamics:
    def __init__(
        self,
        tau_v: float = 120.0,
        tau_a: float = 120.0,
        kappa: float = 0.5,
    ):
        self.tau = np.array([tau_v, tau_a])
        self.kappa = kappa
        self.state = np.array([0.0, 0.0])  # [v, a]
        self.last_update = time.time()
    
    def update(self, delta_v: float, delta_a: float) -> Tuple[float, float]:
        """应用增量并衰减"""
        now = time.time()
        dt = now - self.last_update
        
        decay = np.exp(-dt / self.tau)
        delta = np.array([delta_v, delta_a])
        
        self.state = self.state * decay + self.kappa * delta
        self.state = np.clip(self.state, -1.0, 1.0)
        self.last_update = now
        
        return (float(self.state[0]), float(self.state[1]))
    
    def tick_only(self) -> Tuple[float, float]:
        """仅衰减，无新输入（用于周期性回归）"""
        return self.update(0.0, 0.0)
    
    def get_state(self) -> Tuple[float, float]:
        return (float(self.state[0]), float(self.state[1]))


# 向量 -> 文本映射表
VA_TO_TEXT = {
    (0.6, 0.6): "感到很开心很兴奋",
    (0.5, 0.3): "感到愉快",
    (0.5, 0.0): "感到平和",
    (0.3, -0.3): "感到放松",
    (0.0, 0.0): "感觉很平静",
    (-0.2, 0.2): "有点烦躁",
    (-0.5, 0.4): "感到生气",
    (-0.5, 0.0): "感到低落",
    (-0.4, -0.3): "感到有点沮丧",
}

def va_to_text(v: float, a: float) -> str:
    """找最接近的文本描述"""
    best_dist = float("inf")
    best_text = "感觉很平静"
    for (kv, ka), text in VA_TO_TEXT.items():
        d = abs(v - kv) + abs(a - ka)
        if d < best_dist:
            best_dist = d
            best_text = text
    return best_text
```

## 六、阶段 5：整合为 MoodEngine（混合词典 + LLM）

```python
# src/mood/mood_engine.py（效果优先版核心）
import random
from .mood_estimator import estimate_from_lexicon
from .mood_dynamics import MoodDynamics, va_to_text

class MoodEngine:
    def __init__(
        self,
        chat_id: str,
        lexicon_threshold: float = 0.6,
        use_llm_fallback: bool = True,
        # ... 其他配置
    ):
        self.chat_id = chat_id
        self.lexicon_threshold = lexicon_threshold
        self.use_llm_fallback = use_llm_fallback
        self.dynamics = MoodDynamics(tau_v=120, tau_a=120, kappa=0.5)
        self.mood_text = "感觉很平静"
        self.last_change_time = 0.0
    
    async def _estimate_with_fallback(self, text: str) -> tuple[float, float, str]:
        """
        1. 先词典估计
        2. 置信度低于阈值则调用 LLM
        Returns: (v, a, source)  # source: "lexicon" | "llm"
        """
        v, a, conf = estimate_from_lexicon(text)
        
        if conf >= self.lexicon_threshold:
            return (v, a, "lexicon")
        
        if self.use_llm_fallback:
            # 调用现有 mood_model（与 mood_manager 一致）
            v_llm, a_llm, text_desc = await self._llm_estimate(text)
            return (v_llm, a_llm, "llm")
        
        # 无 LLM 时仍用词典结果，但置信度低
        return (v, a, "lexicon")
    
    async def observe_message(self, message_text: str, interested_rate: float = 0.5):
        """收到消息时调用"""
        import time
        now = time.time()
        
        # 概率判断（沿用现有逻辑，简化示例）
        if random.random() > 0.1 * interested_rate:
            return
        
        v_delta, a_delta, source = await self._estimate_with_fallback(message_text)
        v_new, a_new = self.dynamics.update(v_delta, a_delta)
        
        self.mood_text = va_to_text(v_new, a_new)
        self.last_change_time = now
        
        return self.mood_text
    
    def get_mood_for_prompt(self) -> str:
        """供 LLM 生成时使用的 prompt 片段"""
        v, a = self.dynamics.get_state()
        return f"当前情绪：{self.mood_text}（valence={v:.2f}, arousal={a:.2f}）"
```

## 九、数据流示意

```

        |
        v

        | 词典: 今天(0.1,0.2), 好(0.3,0.1), 累(-0.4,-0.3)
```

## 十一、总结：效果优先版的特点

| 环节     | 做法                         | 训练/数据需求 |
|----------|------------------------------|----------------|
| 感知     | 词典 + LLM fallback          | 无需训练，仅词典 |
| 动力学   | 指数衰减 + 增量              | 无               |
| 文本映射 | 预定义映射表                 | 可人工微调       |
| 集成     | 保持 ChatMood 接口不变       | 无               |

**全程无模型训练**，仅依赖：
1. NRC-VAD 词典（下载即用）
2. 已有 LLM（mood_model）作为低置信度回退
3. 简单动力学公式与映射表

即可完成「准备 → 加载 → 估计 → 更新 → 使用」的闭环。
